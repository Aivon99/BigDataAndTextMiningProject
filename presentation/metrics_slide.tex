\begin{textblock}{12.1}(0.6,0.38)
  {\rmfamily\bfseries\fontsize{34}{40}\selectfont Evaluation metrics}\par\vspace{6pt}
  {\color{grey}\fontsize{16}{20}\selectfont
  Let $N$ be the number of test examples, $B_i = (b_{i,1}, \dots, b_{i,64})$ the true board and
  $\hat{B}_i$ the predicted one; let $y_i$ and $\hat{y}_i$ be the true and predicted FEN strings.}
\end{textblock}

% ---------- top row: EM, square accuracy, CER ----------
\begin{textblock}{3.9}(0.6,1.9)
\begin{metric}[height=2.35in]{Board exact match}
  \desc{Share of boards predicted entirely correctly.}
  \vfill
  \[
  \mathrm{EM} = \frac{1}{N} \sum_{i=1}^{N} \mathbbm{1}\!\left[\hat{B}_i = B_i\right]
  \]
  \vfill
\end{metric}
\end{textblock}

\begin{textblock}{3.9}(4.715,1.9)
\begin{metric}[height=2.35in]{Square-by-square accuracy}
  \desc{Share of the 64 squares (piece or empty) predicted correctly.}
  \vfill
  \[
  \mathrm{Acc}_{\mathrm{sq}} = \frac{1}{64\,N} \sum_{i=1}^{N} \sum_{s=1}^{64}
  \mathbbm{1}\!\left[\hat{b}_{i,s} = b_{i,s}\right]
  \]
  \vfill
\end{metric}
\end{textblock}

\begin{textblock}{3.9}(8.83,1.9)
\begin{metric}[height=2.35in]{Character Error Rate (CER)}
  \desc{Edit distance between FEN strings, normalised by the true length.}
  \vspace{-4pt}
  \[
  \mathrm{CER} = \frac{1}{N} \sum_{i=1}^{N} \frac{\mathrm{Lev}(\hat{y}_i, y_i)}{|y_i|}
  \]
  \vspace{-6pt}
  {\normalsize For one example, with $S$ substitutions, $D$ deletions and $I$ insertions:}
  \vspace{-6pt}
  \[
  \mathrm{CER} = \frac{S + D + I}{|y|}
  \]
\end{metric}
\end{textblock}

% ---------- bottom row: Levenshtein ----------
\begin{textblock}{12.13}(0.6,4.45)
\begin{metric}[height=2.3in]{Levenshtein distance}
  \desc{Minimum number of single-character edits that turn $\hat{y}$ into $y$. With $a = \hat{y}$ and $b = y$:}
  \vfill
  \begin{minipage}[c]{8.3in}
  \[
  \mathrm{Lev}_{a,b}(i,j) =
  \begin{cases}
  \max(i,j) & \text{if } \min(i,j) = 0, \\[6pt]
  \min
  \left\{
  \begin{array}{l}
  \mathrm{Lev}_{a,b}(i-1,j) + 1 \\
  \mathrm{Lev}_{a,b}(i,j-1) + 1 \\
  \mathrm{Lev}_{a,b}(i-1,j-1) + \mathbbm{1}\!\left[a_i \neq b_j\right]
  \end{array}
  \right.
  & \text{otherwise}
  \end{cases}
  \]
  \end{minipage}\hfill
  \begin{minipage}[c]{3.2in}
  The final value is\par\vspace{4pt}
  $\mathrm{Lev}(a,b) = \mathrm{Lev}_{a,b}(|a|,|b|)$
  \end{minipage}
  \vfill
\end{metric}
\end{textblock}
